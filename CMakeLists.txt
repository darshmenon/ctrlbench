cmake_minimum_required(VERSION 3.16)
project(ctrlbench LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")

find_package(Eigen3 REQUIRED)

# MuJoCo C library: pass -DMUJOCO_DIR=<dir with include/ and libmujoco.so.*> (the pip package directory works)
if(MUJOCO_DIR)
  file(GLOB MUJOCO_LIB "${MUJOCO_DIR}/libmujoco.so*")
  list(GET MUJOCO_LIB 0 MUJOCO_LIB)
  add_executable(ik_mj cpp/ik_mj.cpp)
  target_include_directories(ik_mj PRIVATE ${MUJOCO_DIR}/include)
  target_link_libraries(ik_mj Eigen3::Eigen ${MUJOCO_LIB})
  set_target_properties(ik_mj PROPERTIES BUILD_RPATH ${MUJOCO_DIR})
endif()

add_executable(ik_dh cpp/ik_dh.cpp)
target_link_libraries(ik_dh Eigen3::Eigen)
add_executable(timer_overhead cpp/timer_overhead.cpp)
target_link_libraries(timer_overhead Eigen3::Eigen)

find_package(pinocchio QUIET)
if(pinocchio_FOUND)
  add_executable(imp cpp/imp.cpp)
  target_link_libraries(imp pinocchio::pinocchio)
else()
  message(WARNING "pinocchio not found: the impedance benchmarks are skipped")
endif()

find_package(rclcpp QUIET)
find_package(sensor_msgs QUIET)
find_package(std_msgs QUIET)
if(rclcpp_FOUND AND sensor_msgs_FOUND AND std_msgs_FOUND)
  foreach(t rcl_dut rcl_harness)
    add_executable(${t} cpp/${t}.cpp)
    target_link_libraries(${t} rclcpp::rclcpp ${sensor_msgs_TARGETS} ${std_msgs_TARGETS})
  endforeach()
else()
  message(WARNING "rclcpp not found (source /opt/ros/humble/setup.bash): the ROS 2 benchmark is skipped")
endif()
